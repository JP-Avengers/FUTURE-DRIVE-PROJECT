## 명령 (11.8, Windows Git Bash, torch 없이 · `PYTHONIOENCODING=utf-8`)

```bash
ZS=~/Downloads/xm_ft_aw30_kaggle/eval_zeroshot_test                 # 제로샷 · v1.0 test
V01=~/Downloads/xm_ft_kaggle_full/xm_ft_kaggle_e10/eval_test
S1=~/Downloads/xm_ft_seed1_kaggle/xm_ft_v0.1_seed1_e10/eval_test    # v0.1 설정 + --seed 1

python e2e/steer_diag.py ZS=$ZS/frames.csv v0.1=$V01/frames.csv seed1=$S1/frames.csv p3=$P3/frames.csv -o $O/steer_diag.md
python e2e/rollout_drift.py ZS=$ZS/frames.csv v0.1=$V01/frames.csv seed1=$S1/frames.csv p3=$P3/frames.csv -o $O/rollout_drift.md
python e2e/fill_report.py --train $P3R --zs $ZS --ft $P3 --ckpt "$P3R/checkpoints/epoch=9-step=11910.ckpt" --version xm_ft_v0.1_p3 --out $O/report_tables.md   # ckpt 파일 없음 → 【sha12】 만 25b8ab2106af 로 손수

python e2e/weak_segments.py --a $V01 --b $P3 -o <임시> --top 8    # → weak_segments_vs_v0.1.md
```
